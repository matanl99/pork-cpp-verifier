# pork: sammanfattningar av std för lånemodellen (STN-62).
#
# Det här är antaganden, inte bevis: libstdc++ analyseras aldrig. Varje rad
# ska gå att granska mot standarden (paragrafen står i kommentaren). Format
# och betydelse: core/LoanSummaries.hpp. Det som saknas här får den
# försiktiga standarden (ändrar allt icke-const det når, resultatet lånar
# allt), så en saknad rad gör modellen strängare, aldrig osund.

# --- Iteratorer och vyer ([[gsl::Pointer]]) --------------------------------
# [iterator.requirements]: avreferering ger en referens in i det iteratorn
# pekar på; stegning och tilldelning ändrar bara iteratorn själv.
pointer operator* operator-> operator[] : access
pointer operator++ operator-- operator+= operator-= operator= : self
pointer operator== operator!= operator< operator> operator<= operator>= operator<=> : reads
pointer <const> : reads

# --- Behållare och ägande typer ([[gsl::Owner]]) ---------------------------
# Åtkomst lånar element (ägd lagring) utan att ändra behållaren.
owner begin end cbegin cend rbegin rend crbegin crend : access
owner front back data at find lower_bound upper_bound equal_range value : access
owner operator[] operator* operator-> : access
# [map.access], [unord.map.elem]: operator[] sätter in om nyckeln saknas.
# I map flyttas inga noder ([associative.reqmts.general]); unordered_map kan rehasha.
owner std::map::operator[] : insert
owner std::unordered_map::operator[] : writeAccess
owner size empty capacity max_size count contains has_value index length : reads
owner compare c_str starts_with ends_with get_allocator key_comp value_comp : reads
# [optional.observe]: value_or läser *this. Raden säger inget om v: U&& skickas
# vidare till en konvertering T(U) som kan vara användarkod (P-d i
# docs/stone_model.md, STN-25).
owner value_or : reads
owner operator== operator!= operator< operator> operator<= operator>= operator<=> conversion : reads

# Insättning med en position: positionen pekar in i behållaren per definition
# ([sequence.reqmts] a.insert(p, t), a.emplace(p, args)). Bara första
# argumentet: i a.insert(p, i, j) får i och j inte peka in i a.
owner insert(iterator,...) emplace(iterator,...) : default position copies insertsArguments

# Värdet som kopieras in får vara ett element i behållaren: [sequence.reqmts]
# har inget sådant förvillkor för push_back(t), insert(p, t), insert(p, n, t)
# eller resize(n, t) (LWG 526), till skillnad från a.assign(n, t), där t inte
# får referera in i a. Gäller bara const T&-parametrar (inte T&&,
# [res.on.arguments], och inte emplace_back(args&&...)).
owner push_back push_front : default copies insertsArguments
owner resize : default copies

# [associative.reqmts.general]/[list.modifiers]: insättning ogiltigförklarar
# inga iteratorer eller referenser (noder flyttas inte).
owner std::map::insert std::map::emplace std::map::try_emplace : insert insertsArguments
owner std::multimap::insert std::multimap::emplace : insert insertsArguments
owner std::set::insert std::set::emplace : insert insertsArguments
owner std::multiset::insert std::multiset::emplace : insert insertsArguments
owner std::map::insert(iterator,...) std::map::emplace_hint(iterator,...) std::map::try_emplace(iterator,...) : insert position insertsArguments
owner std::multimap::insert(iterator,...) std::multimap::emplace_hint(iterator,...) : insert position insertsArguments
owner std::set::insert(iterator,...) std::set::emplace_hint(iterator,...) : insert position insertsArguments
owner std::multiset::insert(iterator,...) std::multiset::emplace_hint(iterator,...) : insert position insertsArguments
owner std::list::push_back std::list::push_front std::list::emplace_back std::list::emplace_front : insert insertsArguments
owner std::list::insert(iterator,...) std::list::emplace(iterator,...) : insert position insertsArguments

# Radering via iterator: iteratorn får peka in i behållaren och blir ogiltig
# ([sequence.reqmts] a.erase(q), [associative.reqmts] a.erase(q)); den
# returnerade iteratorn ersätter den (`it = c.erase(it)`). Övriga lån in i
# behållaren dör (försiktigt: för noder bara det raderade elementet).
owner erase(iterator) : default position consumes keepsElements
owner erase(iterator,iterator) : default range keepsElements

# --- Tomhet (STN-89) --------------------------------------------------------
# [unique.ptr.single.observers]: operator* och operator-> har förvillkoret
# get() != nullptr; get() ger den lagrade pekaren (null när *this är tom).
owner std::unique_ptr::operator* std::unique_ptr::operator-> : access nonEmpty
owner std::unique_ptr::get : default resultNullIfEmpty
# [unique.ptr.single.modifiers] release(): ger get() och gör den tom (STN-90).
owner std::unique_ptr::release : default resultNullIfEmpty
# [unique.ptr.single.modifiers] swap(u), [utility.swap] swap(a, b): byter innehåll.
owner std::unique_ptr::swap : default swaps
# [sequence.reqmts], [vector.modifiers], [deque.modifiers]: emplace_back(args)
# konstruerar elementet ur args (utan args värdeinitierat: en tom unique_ptr);
# pop_back, pop_front och clear tar bort element, reserve och shrink_to_fit
# flyttar dem ([unique.ptr.single.ctor]: en flytt behåller pekaren).
owner emplace_back emplace_front : default insertsArguments
owner pop_back pop_front clear reserve shrink_to_fit : default keepsElements

# --- Fria funktioner (även std::ranges-algoritmer) -------------------------
free operator== operator!= operator< operator> operator<= operator>= operator<=> : reads

free get_if : access

# returnerar std::forward/std::move av elementet, en referens; inget flyttas

free get : access forwards

free begin end cbegin cend rbegin rend crbegin crend : access
# [unique.ptr.create]: make_unique ger unique_ptr<T>(new T(...)), aldrig tom

free make_unique make_unique_for_overwrite : default resultNonEmpty
free swap : default swaps
# [forward], [utility.as.const]: resultatet är argumentet.
free move forward as_const : identity

free lower_bound upper_bound partition_point min_element max_element : access interval(0,1)

# --- Intervallformer av std::ranges-algoritmerna (STN-87) ------------------
# [algorithms.requirements]: en överlagring med intervallargument gör som den
# med iterator och sentinel, med ranges::begin(r) och ranges::end(r) för varje
# intervall r. Ett intervall är alltså ett eget (first, last): andra iterator-
# argument som kan peka in i det avvisas när funktionen kan skriva
# (ranges::copy(s, s.begin() + 1)). Raderna ovan och nedan utan `range` i
# formen gäller inte intervallformen (inte heller std::move(x):s identity).
# [alg.find], [alg.search], [alg.binary.search], [alg.min.max],
# [alg.partitions], [alg.sort]: resultatet är en position (eller ett delintervall)
# i det första intervallet.
free find(range,...) find_if(range,...) find_if_not(range,...) adjacent_find(range,...) search_n(range,...) : access
free lower_bound(range,...) upper_bound(range,...) equal_range(range,...) partition_point(range,...) : access
free min_element(range,...) max_element(range,...) is_sorted_until(range,...) is_heap_until(range,...) : access
free find_end(range,range,...) find_first_of(range,range,...) search(range,range,...) : access
# [alg.remove], [alg.unique]: resultatet är {i, last}, ett delintervall av r (STN-85).
free remove(range,...) remove_if(range,...) unique(range,...) : default resultIn(0)
# Bara roller; effekten är den försiktiga (default). [alg.sort], [alg.fill],
# [alg.reverse], [alg.foreach], [alg.count], [alg.replace], [alg.all.of],
# [alg.is.sorted], [alg.binary.search], [alg.copy], [alg.move], [alg.transform],
# [alg.equal].
free sort(range,...) stable_sort(range,...) fill(range,...) reverse(range,...) for_each(range,...) : default
free count(range,...) count_if(range,...) replace(range,...) replace_if(range,...) : default
free all_of(range,...) any_of(range,...) none_of(range,...) is_sorted(range,...) binary_search(range,...) : default
free copy(range,...) copy_if(range,...) copy_backward(range,...) move(range,...) move_backward(range,...) : default
free transform(range,...) transform(range,range,...) equal(range,range,...) : default

# --- Argumentroller för övriga algoritmer (STN-77) --------------------------
# interval(i,j): [first, last) i samma objekt ([iterator.requirements.general]:
# [i, j) är ett giltigt intervall bara om j kan nås från i). Raderna ger bara
# roller; effekten är den försiktiga (default). Andra iteratorargument som
# kan peka in i samma objekt avvisas när funktionen kan skriva genom dem.
# [alg.sort], [alg.fill], [alg.reverse], [alg.foreach], [alg.count],
# [alg.replace], [alg.remove], [alg.unique], [alg.all.of], [alg.is.sorted],
# [numeric.iota], [accumulate], [iterator.operations] (distance).
free sort stable_sort fill reverse for_each count count_if replace replace_if : default interval(0,1)
free iota all_of any_of none_of is_sorted accumulate distance : default interval(0,1)
# [alg.remove], [alg.unique]: "Returns: The end of the resulting range", en position i
# [first, last]. Storleken på behållaren ändras inte (STN-85, erase-remove).
free remove remove_if unique : default interval(0,1) resultIn(0)
# [alg.copy], [alg.move], [alg.transform]: result ligger inte i [first, last).
free copy copy_if copy_backward move_backward transform : default interval(0,1)
# [alg.transform] ranges::transform(first1, last1, first2, last2, result, op):
# två intervall. std::transform(first1, last1, first2, result, op) har fyra
# iteratorparametrar, first2 är där ensam (raden ovan).
free transform(iterator,iterator,iterator,iterator,iterator,...) : default interval(0,1) interval(2,3)
# Algoritmen std::move(first, last, result), inte std::move(x) ([forward]).
free move(iterator,...) : default interval(0,1)
# [alg.equal], [alg.swap]: två intervall (swap_ranges: first2 ensam).
free equal swap_ranges : default interval(0,1) interval(2,3)
# [alg.rotate], [alg.nth.element], [partial.sort], [alg.merge]: (first, middle, last).
free rotate nth_element partial_sort inplace_merge : default interval(0,1) interval(1,2) interval(0,2)
# [alg.copy] copy_n(first, n, result), [alg.fill] fill_n, [alg.swap] iter_swap(a, b):
# inga intervall, varje iterator är en egen position.
free copy_n fill_n iter_swap : default
